# IPlugControls standalone APP, P3b: the APP gets keys (text entry, the about box's Escape), clipboard, locked
# cursor. No REAPER here, so values are checked on screen. View-relative coordinates as in iplugcontrols-p3b.txt.
alive __PROC__
shot a00-start
# cursors from the Xcursor theme: the corner resizer's diagonal one, the default elsewhere
move 1016 760
wait 0.5
expect-cursor nwse-resize
move 500 400
wait 0.5
expect-cursor default
# the knob hides and locks the cursor while dragging
drag-check 322 237 322 330 expect-cursor blank
wait 0.5
expect-cursor default
shot a01-knob-drag
expect-change a00-start a01-knob-drag 262 170 120 130
expect-pointer 322 237
# IGraphics text entry: typed text commits with Return
dclick 941 437
wait 1
shot a02-textentry-open
type 42
key Return
wait 0.5
shot a03-textentry-committed
expect-change a02-textentry-open a03-textentry-committed 870 410 110 40
# Escape cancels, and does not reach SWELL (whose Escape would close the APP's dialog)
dclick 941 437
wait 1
type 13
key Escape
wait 0.5
shot a04-textentry-cancelled
expect-same a03-textentry-committed a04-textentry-cancelled 870 410 110 40
alive __PROC__
# clipboard both ways
set-clipboard 55
dclick 941 437
wait 1
key ctrl+v
key Return
wait 0.5
shot a05-pasted
expect-change a04-textentry-cancelled a05-pasted 870 410 110 40
dclick 941 437
wait 1
key ctrl+c
key Escape
wait 0.5
expect-clipboard 55
# stand-in ("OS") text entry and menu
click 224 692
click 224 720
wait 0.5
click 828 284
wait 1
shot a06-popup-os
expect-change a05-pasted a06-popup-os 760 270 200 150
click 833 358
wait 1
shot a07-popup-chosen
dclick 941 437
wait 1
type 7
key Return
wait 1
shot a08-textentry-os
expect-change a07-popup-chosen a08-textentry-os 870 410 110 40
alive __PROC__
# right-click Paste in stand-in edits, twice: the edit menu must belong to the current text entry control, not to
# the first one ever right-clicked (a function-local static in ITextEntryControl did, a use-after-free here)
set-clipboard 33
dclick 941 437
wait 1
rclick 941 437
wait 1
click 945 481
wait 0.5
key Return
wait 0.5
shot a08b-pasted-33
set-clipboard 44
dclick 941 437
wait 1
rclick 941 437
wait 1
click 945 481
wait 0.5
key Return
wait 0.5
shot a08c-pasted-44
expect-change a08b-pasted-33 a08c-pasted-44 870 410 110 40
# a key outside text entry: the editor has the focus since the last click (APP only), so the about box gets Escape
click 993 737
wait 1
shot a09-about
key Escape
wait 1
shot a10-about-dismissed
expect-change a09-about a10-about-dismissed 0 0 1024 768
alive __PROC__
