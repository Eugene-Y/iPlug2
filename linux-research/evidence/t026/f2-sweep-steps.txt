# T-026 F2: fast CC sweeps from a MIDI item (CC1 every ~2 ms, faster than Sand's 20 ms idle tick) after a
# CC learn on the comp knob (same menu path as T-024's f2-steps.txt). The diagnostic build logs every
# comp-amount change with its source (4 = MIDI CC, 1 = host) to REAPER's stderr.
shot f00-start
[CLAP] rclick 256 94
[VST3] rclick-hold 256 94
wait 1.5
[CLAP] move 230 81
[VST3] move 311 240
wait 1.5
[CLAP] click 373 21
[VST3] click 545 244
wait 0.5
lua-expect armed local t=reaper.GetTrack(0,0); reaper.SetMediaTrackInfo_Value(t,"I_RECINPUT",4096+63*32); reaper.SetMediaTrackInfo_Value(t,"I_RECMON",1); reaper.SetMediaTrackInfo_Value(t,"I_RECARM",1); out("armed")
wait 1
lua-expect sent reaper.StuffMIDIMessage(0,0xB0,1,64); out("sent")
wait 1
lua-expect disarmed local t=reaper.GetTrack(0,0); reaper.SetMediaTrackInfo_Value(t,"I_RECARM",0); reaper.SetMediaTrackInfo_Value(t,"I_RECMON",0); out("disarmed")
# sweeps at 0.5 s (127->0 in 200 ms), 1.0 s (0->127 in 100 ms), 1.5 s (127->0 in 50 ms), 2.0 s (0->127 in 20 ms)
lua-expect item.ccs.[0-9]+ local t=reaper.GetTrack(0,0); local it=reaper.CreateNewMIDIItemInProj(t,0,3); local tk=reaper.GetActiveTake(it); local n=0; local function ramp(t0,len,a,b) local steps=math.max(2,math.floor(len/0.002)); for k=0,steps do local v=math.floor(a+(b-a)*k/steps+0.5); reaper.MIDI_InsertCC(tk,false,false,reaper.MIDI_GetPPQPosFromProjTime(tk,t0+len*k/steps),0xB0,0,1,v); n=n+1 end end; ramp(0.5,0.2,127,0); ramp(1.0,0.1,0,127); ramp(1.5,0.05,127,0); ramp(2.0,0.02,0,127); reaper.MIDI_Sort(tk); out("item","ccs",n)
lua-expect mode0 local t=reaper.GetTrack(0,0); reaper.SetMediaTrackInfo_Value(t,"I_AUTOMODE",0); reaper.SetEditCurPos(0,false,false); reaper.OnPlayButton(); out("mode0")
wait 3.5
lua-expect stopped reaper.OnStopButton(); io.stderr:write("T026DIAG pass-end mode0\n"); out("stopped")
wait 1
lua-expect mode3 local t=reaper.GetTrack(0,0); reaper.SetMediaTrackInfo_Value(t,"I_AUTOMODE",3); reaper.SetEditCurPos(0,false,false); reaper.OnPlayButton(); out("mode3")
wait 3.5
lua-expect stopped reaper.OnStopButton(); io.stderr:write("T026DIAG pass-end mode3\n"); out("stopped")
wait 1
lua-expect host-value local t=reaper.GetTrack(0,0); for i=0,reaper.TrackFX_GetNumParams(t,0)-1 do local _,n=reaper.TrackFX_GetParamName(t,0,i,""); if n=="compression amnt" then local _,v=reaper.TrackFX_GetFormattedParamValue(t,0,i,""); local e=reaper.GetFXEnvelope(t,0,i,false); out("host-value", n, v, "env-points", e and reaper.CountEnvelopePoints(e) or "none") end end
shot f01-end
alive __PROC__
